cmake_minimum_required(VERSION 3.22)
project(nhl07_android C CXX)
option(NHL07_CUSTOM_GRAPHICS "Build optional ANGLE/Adreno driver loader" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Use the Android NDK, arm64-v8a")
endif()
set(BACKEND "${CMAKE_CURRENT_SOURCE_DIR}/vendor/xboxrecomp")
set(GAME "${CMAKE_CURRENT_SOURCE_DIR}/../NHL07-x64")
add_compile_definitions(RECOMP_NATIVE_API_ONLY=1)
add_compile_options(-fno-strict-aliasing -ffp-contract=off -fwrapv -Wno-parentheses-equality)
include_directories("${GAME}/src/recomp/gen" "${BACKEND}/templates/runtime" "${BACKEND}/src"
  "${GAME}/src"
  "${BACKEND}/src/kernel" "${BACKEND}/src/platform" "${BACKEND}/include" "${BACKEND}/src/d3d")
file(GLOB KERNEL "${BACKEND}/src/kernel/*.c")
list(FILTER KERNEL EXCLUDE REGEX "/nv2a_[^/]*\\.c$")
add_library(nhl07_kernel STATIC ${KERNEL} "${BACKEND}/src/platform/win32_compat.c")
target_compile_options(nhl07_kernel PRIVATE -Wno-incompatible-function-pointer-types -Wno-pointer-sign)
file(GLOB TRANSLATED CONFIGURE_DEPENDS "${GAME}/src/recomp/gen/*.c")
add_library(nhl07_game STATIC ${TRANSLATED} "${GAME}/src/overrides.c" "${GAME}/src/recomp_manual.c")
target_sources(nhl07_game PRIVATE "${GAME}/src/runtime_callbacks.c")
target_sources(nhl07_game PRIVATE "${GAME}/src/static_recovered.c")
target_sources(nhl07_game PRIVATE "${GAME}/src/recovered_thunks.c")
target_sources(nhl07_game PRIVATE "${GAME}/src/native_graphics_hooks.c")
target_sources(nhl07_game PRIVATE "${GAME}/src/native_resource_cache.c")
target_sources(nhl07_game PRIVATE "${GAME}/src/native_resource_lifetime.c")
target_sources(nhl07_game PRIVATE src/native_text_hooks.c)
target_compile_options(nhl07_game PRIVATE -Wno-unused-variable -Wno-shift-count-overflow)
add_library(nhl07_core SHARED src/native_android.c src/path_case.c src/native_graphics.c src/android_activity.c src/probe_main.c)
target_sources(nhl07_core PRIVATE src/asset_overlays.c)
target_sources(nhl07_core PRIVATE src/native_fps.c src/native_pause.c)
target_sources(nhl07_core PRIVATE src/native_frame_pacer.c)
target_sources(nhl07_core PRIVATE src/native_input.c src/native_input_hooks.c src/native_audio.c src/native_audio_hooks.c)
target_sources(nhl07_core PRIVATE src/native_dsp.c src/native_voice.c src/xbox_adpcm.c)
# Keep the integer DSP interpreter local to this shared object and allow its
# arithmetic / addressing helpers to optimize independently of the game code.
# No fast-math or changes to fixed-point precision / effect processing.
set_property(SOURCE src/native_dsp.c APPEND PROPERTY COMPILE_OPTIONS -O3 -fvisibility=hidden)
target_sources(nhl07_core PRIVATE src/nv2a_draw.cpp src/shaders/shader_compile.cpp
  src/shaders/shader_vertex.c src/shaders/shader_pixel.c src/shaders/texture_rgba.cpp src/shaders/d3d_texture_model.c)
target_sources(nhl07_core PRIVATE src/render_targets.cpp src/render_target_pixels.cpp)
target_sources(nhl07_core PRIVATE src/texture_cache.cpp)
target_sources(nhl07_core PRIVATE src/render_profile.cpp)
set_property(SOURCE src/render_profile.cpp APPEND PROPERTY
  COMPILE_DEFINITIONS NHL07_DEFAULT_SHADER_PREP_CACHE=0 NHL07_DEFAULT_EXACT_BUFFER_CACHE=1)
target_sources(nhl07_core PRIVATE src/scaled_upload.cpp)
target_sources(nhl07_core PRIVATE src/stream_upload.cpp src/exact_buffer_cache.cpp)
if(NHL07_CUSTOM_GRAPHICS)
target_sources(nhl07_core PRIVATE src/angle_backend.cpp src/native_gl_dispatch.cpp)
set_source_files_properties(src/native_graphics.c src/nv2a_draw.cpp src/render_targets.cpp
  src/scaled_upload.cpp src/stream_upload.cpp src/exact_buffer_cache.cpp src/texture_cache.cpp src/shaders/shader_compile.cpp
  PROPERTIES COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/src/native_gl_dispatch.h")
set_source_files_properties(src/angle_backend.cpp src/native_gl_dispatch.cpp PROPERTIES
  COMPILE_DEFINITIONS NHL07_GL_DISPATCH_IMPLEMENTATION=1)
else()
target_sources(nhl07_core PRIVATE src/system_graphics_backend.c)
endif()
target_include_directories(nhl07_core PRIVATE src/shaders deps/glslang deps/spirv-cross deps/dsp56300 deps/adpcm-xq)
foreach(lib IN ITEMS glslang glslang-default-resource-limits SPIRV spirv-cross-core spirv-cross-glsl)
  add_library(${lib} STATIC IMPORTED)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/deps/shader-libs/lib${lib}.a")
endforeach()
target_link_libraries(nhl07_core PRIVATE glslang glslang-default-resource-limits SPIRV spirv-cross-glsl spirv-cross-core)
target_compile_definitions(nhl07_core PRIVATE NHL07_PROBE_HANDLERS_ONLY=1)
target_link_libraries(nhl07_core PRIVATE nhl07_game nhl07_kernel m log dl EGL GLESv3 android aaudio)
target_link_options(nhl07_core PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
add_executable(nhl07_probe src/probe_main.c)
target_link_libraries(nhl07_probe PRIVATE nhl07_core)
target_link_options(nhl07_probe PRIVATE -Wl,-z,max-page-size=16384 -Wl,-rpath,\$ORIGIN)
